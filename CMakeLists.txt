cmake_minimum_required(VERSION 3.24)
project(tintero LANGUAGES CXX)

foreach(option ENABLE_HIP ENABLE_CUDA BUILD_TESTS)
  if(DEFINED MAYA_${option})
    set(STRATA_${option} "${MAYA_${option}}")
  endif()
endforeach()
option(STRATA_ENABLE_HIP "Build for AMD RDNA 3 and 3.5" OFF)
option(STRATA_ENABLE_CUDA "Build for NVIDIA Ampere or newer" OFF)
option(BUILD_TESTING "Build numerical reference tests" ON)
option(STRATA_BUILD_TESTS "Build numerical reference tests" ${BUILD_TESTING})

if(STRATA_ENABLE_HIP AND STRATA_ENABLE_CUDA)
  message(FATAL_ERROR "Select one GPU backend per build directory")
elseif(NOT STRATA_ENABLE_HIP AND NOT STRATA_ENABLE_CUDA)
  return()
endif()

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64|ARM64)$")
  add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:-ffp-contract=off>")
endif()
find_package(Threads REQUIRED)

set(ik2_kernel_sources
  src/ik2/ik2_kernels.cu
  src/ik2/ik2_attention_kernels.cu
  src/ik2/ik2_draft_kernels.cu
  src/ik2/ik2_matmul_kernels.cu
  src/ik2/ik2_sampling_kernels.cu
)

set(ik2_engine_sources
  src/ik2/ik2_main.cu
  src/ik2/ik2_prefill.cu
  src/ik2/ik2_decode.cu
)

if(STRATA_ENABLE_HIP)
  set(CMAKE_HIP_ARCHITECTURES "gfx1100;gfx1151" CACHE STRING "HIP targets")
  foreach(arch IN LISTS CMAKE_HIP_ARCHITECTURES)
    if(NOT arch MATCHES "^gfx(110[0-3]|115[01])$")
      message(FATAL_ERROR "Unsupported HIP target '${arch}'. Select an RDNA 3 or 3.5 Wave32 target.")
    endif()
  endforeach()
  enable_language(HIP)
  set(CMAKE_HIP_STANDARD 20)
  find_package(hip CONFIG REQUIRED)
  find_package(hipblas CONFIG REQUIRED)
  set(CMAKE_POSITION_INDEPENDENT_CODE ON)
  set_source_files_properties(${ik2_kernel_sources} ${ik2_engine_sources}
    src/ik2/ik2_draft.cu tests/ik2_kernels_test.cu tests/ik2_dflash2_test.cu PROPERTIES LANGUAGE HIP)
else()
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    if(DEFINED ENV{CUDAARCHS})
      set(CMAKE_CUDA_ARCHITECTURES "$ENV{CUDAARCHS}")
    else()
      set(CMAKE_CUDA_ARCHITECTURES 80-real CACHE STRING "CUDA architectures")
    endif()
  endif()
  # Older CMake releases accept numeric architectures during compiler discovery.
  # Restore the requested architecture-specific feature set on the actual build targets below.
  set(requested_archs "${CMAKE_CUDA_ARCHITECTURES}")
  string(REGEX REPLACE "(90|120|121)a" "\\1" CMAKE_CUDA_ARCHITECTURES "${requested_archs}")
  enable_language(CUDA)
  set(CMAKE_CUDA_STANDARD 17)
  find_package(CUDAToolkit REQUIRED)
  set(CMAKE_CUDA_ARCHITECTURES "${requested_archs}")
  if(CMAKE_CUDA_ARCHITECTURES STREQUAL "native")
    set(CMAKE_CUDA_ARCHITECTURES "${CMAKE_CUDA_ARCHITECTURES_NATIVE}")
  endif()
  set(archs "")
  foreach(arch IN LISTS CMAKE_CUDA_ARCHITECTURES)
    if(NOT arch MATCHES "^([0-9]+)(a?)(-(real|virtual))?$" OR CMAKE_MATCH_1 LESS 80)
      message(FATAL_ERROR "Select explicit CUDA architectures >= 80, got '${arch}'")
    endif()
    string(REGEX REPLACE "^(90|120|121)($|-)" "\\1a\\2" arch "${arch}")
    list(APPEND archs "${arch}")
  endforeach()
  if(NOT archs)
    message(FATAL_ERROR "No native CUDA device detected. Set CMAKE_CUDA_ARCHITECTURES explicitly.")
  endif()
  set(CMAKE_CUDA_ARCHITECTURES "${archs}")
endif()

add_library(ik2_kernels STATIC ${ik2_kernel_sources})
target_include_directories(ik2_kernels PUBLIC src/ik2)
add_library(ik2_draft STATIC src/ik2/ik2_draft.cu)
target_link_libraries(ik2_draft PUBLIC ik2_kernels)
add_executable(tintero ${ik2_engine_sources} src/ik2/ik2_cpu.cpp)
target_link_libraries(tintero PRIVATE ik2_draft Threads::Threads)

if(STRATA_ENABLE_HIP)
  set(compat "${CMAKE_CURRENT_SOURCE_DIR}/include/strata/hip_compat")
  target_sources(ik2_kernels PRIVATE src/ik2/ik2_rdna3.hip)
  target_include_directories(ik2_kernels PUBLIC "${compat}")
  target_compile_options(ik2_kernels PUBLIC "SHELL:-include ${compat}/cuda_runtime.h")
  target_compile_options(ik2_kernels PRIVATE "$<$<COMPILE_LANGUAGE:HIP>:-ffp-contract=on>")
  target_compile_definitions(ik2_kernels PUBLIC STRATA_USE_HIP=1)
  target_link_libraries(ik2_kernels PUBLIC hip::host)
  target_link_libraries(ik2_draft PUBLIC roc::hipblas)
else()
  target_link_libraries(ik2_kernels PUBLIC CUDA::cudart)
  target_link_libraries(ik2_draft PUBLIC CUDA::cublas)
endif()

if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$" AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  find_package(OpenMP QUIET COMPONENTS CXX)
  if(OpenMP_CXX_FOUND)
    target_sources(tintero PRIVATE src/ik2/ik2_cpu_deep.cpp)
    set_source_files_properties(src/ik2/ik2_cpu_deep.cpp PROPERTIES COMPILE_OPTIONS "-mavx2;-mfma;-mf16c;-ffp-contract=off")
    target_compile_definitions(tintero PRIVATE IK2_HOST_AVX2=1)
    target_link_libraries(tintero PRIVATE OpenMP::OpenMP_CXX)
  endif()
endif()

if(STRATA_BUILD_TESTS)
  add_executable(ik2_served_policy_test tests/ik2_served_policy_test.cpp)
  target_include_directories(ik2_served_policy_test PRIVATE src/ik2)
  add_test(NAME ik2_served_policy_test COMMAND ik2_served_policy_test)
  enable_testing()
  add_executable(ik2_host_ref_test tests/ik2_host_ref_test.cpp src/ik2/ik2_cpu.cpp)
  target_link_libraries(ik2_host_ref_test PRIVATE ik2_kernels Threads::Threads)
  add_executable(ik2_dflash2_test tests/ik2_dflash2_test.cu)
  target_link_libraries(ik2_dflash2_test PRIVATE ik2_draft)
  add_test(NAME ik2_dflash2_test COMMAND ik2_dflash2_test)
  add_executable(ik2_kernels_test tests/ik2_kernels_test.cu)
  target_link_libraries(ik2_kernels_test PRIVATE ik2_kernels)
  add_test(NAME ik2_kernels_test COMMAND ik2_kernels_test)
  if(STRATA_ENABLE_CUDA)
    add_test(NAME ik2_batch_group_mma COMMAND ik2_kernels_test --batch-mma)
    add_test(NAME ik2_blackwell_nvfp4 COMMAND ik2_kernels_test --blackwell)
    set_tests_properties(ik2_blackwell_nvfp4 PROPERTIES SKIP_RETURN_CODE 77)
  endif()
  set_tests_properties(ik2_kernels_test PROPERTIES TIMEOUT 300)
endif()
